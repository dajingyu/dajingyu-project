找出数组的出现最多的数字 ​

  • 使用哈希表（字典）
  • 遍历数组，使用哈希表（在JavaScript中是对象或Map）记录每个数字出现的次数。 遍历哈希表，找出出现次数最多的数字。
javascript
function findMostFrequent(arr) {
    let counts = {};
    let maxCount = 0;
    let mostFrequent;

    for(let num of arr) {
        counts[num] = (counts[num] || 0) + 1;
        if(counts[num] > maxCount) {
            maxCount = counts[num];
            mostFrequent = num;
        }
    }

    return mostFrequent;
}

let arr = [2, 2, 1, 1, 1, 2, 2];
console.log(findMostFrequent(arr)); // 输出：2
  • 排序
    • 先将数组排序。
    • 遍历排序后的数组，计数连续相同的数字，比较并记录最大计数值及其对应的数字。
javascript
function findMostFrequentBySort(arr) {
    arr.sort((a, b) => a - b);
    let maxCount = 1;
    let currentCount = 1;
    let mostFrequent = arr[0];

    for(let i = 1; i < arr.length; i++) {
        if(arr[i] === arr[i - 1]) {
            currentCount++;
        } else {
            if(currentCount > maxCount) {
                maxCount = currentCount;
                mostFrequent = arr[i - 1];
            }
            currentCount = 1;
        }
    }

    // 检查最后一个元素的计数
    if(currentCount > maxCount) {
        mostFrequent = arr[arr.length - 1];
    }

    return mostFrequent;
}

let arr = [2, 2, 1, 1, 1, 2, 2];
console.log(findMostFrequentBySort(arr)); // 输出：2

function isUnique(s) {
    const seen = {};
    for (const char of s) {
        if (seen[char]) return false;
        seen[char] = true;
    }
    return true;
}
// 时间复杂度O(n)，空间复杂度O(n)

判断是否为变位词（重新排列字符串） ​

方法1：排序比较法 ​

javascript
function isAnagram(s1, s2) {
    // 长度不同直接返回false
    if (s1.length !== s2.length) return false;
    
    // 排序后比较字符串
    return [...s1].sort().join('') === [...s2].sort().join('');
}
// 时间复杂度：O(n log n) 空间复杂度：O(n)

方法2：哈希表计数法 ​

javascript
function isAnagram(s1, s2) {
    if (s1.length !== s2.length) return false;
    
    const charCount = new Map();
    
    // 统计s1字符出现次数
    for (const char of s1) {
        charCount.set(char, (charCount.get(char) || 0) + 1);
    }
    
    // 验证s2字符
    for (const char of s2) {
        if (!charCount.has(char)) return false;
        charCount.set(char, charCount.get(char) - 1);
        if (charCount.get(char) === 0) charCount.delete(char);
    }
    
    return charCount.size === 0;
}
// 时间复杂度：O(n) 空间复杂度：O(1)（因为字符集固定）

方法3：数组计数法（最优解） ​

javascript
function isAnagram(s1, s2) {
    if (s1.length !== s2.length) return false;
    
    const count = new Array(26).fill(0);
    const base = 'a'.charCodeAt(0);
    
    // 同时统计两个字符串
    for (let i = 0; i < s1.length; i++) {
        count[s1.charCodeAt(i) - base]++; // s1字符计数加1
        count[s2.charCodeAt(i) - base]--; // s2字符计数减1
    }
    
    // 检查所有计数是否为0
    return count.every(c => c === 0);
}
// 时间复杂度：O(n) 空间复杂度：O(1)

方法对比 ​

方法时间复杂度空间复杂度适用场景
排序比较法O(n logn)O(n)快速实现、代码简洁
哈希表计数法O(n)O(1)通用字符集（如包含Unicode）
数组计数法O(n)O(1)仅限小写字母场景（最优解）

测试用例：

javascript

console.log(isAnagram("abc", "bad")); // false
console.log(isAnagram("anagram", "nagaram")); // true

数组原型对象的最后一个元素 ​

  • 编写一段代码实现一个数组方法，使任何数组都可以调用 array.last() 方法，这个方法将返回数组最后一个元素。如果数组中没有元素，则返回 -1 。
  • 算法步骤
    • 在名为 last 的数组原型上定义一个新方法。
    • 在这个方法中，检查数组是否为空。如果是，返回 -1。
    • 如果数组不为空，则返回数组的最后一个元素。最后一个元素可以通过以下方式访问：this[this.length - 1]。

方式一：常规 if 检查

js
Array.prototype.last = function() {
  if (this.length === 0) {
    return -1;
  }

  return this[this.length - 1];
}

方式二：三元运算符

js
Array.prototype.last = function() {
 return this.length === 0 ? -1 ：this[this.length -1]
}

方式三：合并运算符

js
Array.prototype.last = function() {
 return this[this.length -1] ?? -1
}

方式四：pop方法

js
Array.prototype.last = function() {
  let val = this.pop();
  return val !== undefined ? val : -1;
}

根据ID合并数组 ​

现给定两个数组 arr1 和 arr2 ，返回一个新的数组 joinedArray 。两个输入数组中的每个对象都包含一个 id 字段。

joinedArray 是一个通过 id 将 arr1 和 arr2 连接而成的数组。joinedArray 的长度应为唯一值 id 的长度。返回的数组应按 id 升序 排序。

如果一个 id 存在于一个数组中但不存在于另一个数组中，则该对象应包含在结果数组中且不进行修改。

如果两个对象共享一个 id ，则它们的属性应进行合并：

如果一个键只存在于一个对象中，则该键值对应该包含在对象中。 如果一个键在两个对象中都包含，则 arr2 中的值应覆盖 arr1 中的值。

示例 1：

输入： arr1 = [ {"id": 1, "x": 1}, {"id": 2, "x": 9} ], arr2 = [ {"id": 3, "x": 5} ] 输出： [ {"id": 1, "x": 1}, {"id": 2, "x": 9}, {"id": 3, "x": 5} ] 解释：没有共同的 id，因此将 arr1 与 arr2 简单地连接起来。 示例 2：

输入： arr1 = [ {"id": 1, "x": 2, "y": 3}, {"id": 2, "x": 3, "y": 6} ], arr2 = [ {"id": 2, "x": 10, "y": 20}, {"id": 3, "x": 0, "y": 0} ] 输出： [ {"id": 1, "x": 2, "y": 3}, {"id": 2, "x": 10, "y": 20}, {"id": 3, "x": 0, "y": 0} ] 解释：id 为 1 和 id 为 3 的对象在结果数组中保持不变。id 为 2 的两个对象合并在一起。arr2 中的键覆盖 arr1 中的值。 示例 3：

输入： arr1 = [ {"id": 1, "b": {"b": 94},"v": [4, 3], "y": 48} ] arr2 = [ {"id": 1, "b": {"c": 84}, "v": [1, 3]} ] 输出： [ {"id": 1, "b": {"c": 84}, "v": [1, 3], "y": 48} ] 解释：具有 id 为 1 的对象合并在一起。对于键 "b" 和 "v" ，使用 arr2 中的值。由于键 "y" 只存在于 arr1 中，因此取 arr1 的值。

两数之和 ​

给定一个整数数组 nums 和一个目标值 target，请你在该数组中找出和为目标值的那两个整数，并返回它们的数组下标。

方法1：暴力法 ​

javascript
function twoSum(nums, target) {
    for (let i = 0; i < nums.length; i++) {
        for (let j = i + 1; j < nums.length; j++) {
            if (nums[i] + nums[j] === target) {
                return [i, j];
            }
        }
    }
    return [];
}
// 时间复杂度：O(n²) 空间复杂度：O(1)

方法2：哈希表（推荐） ​

javascript
function twoSum(nums, target) {
    const map = new Map();
    for (let i = 0; i < nums.length; i++) {
        const complement = target - nums[i];
        if (map.has(complement)) {
            return [map.get(complement), i];
        }
        map.set(nums[i], i);
    }
    return [];
}
// 时间复杂度：O(n) 空间复杂度：O(n)

// 测试
console.log(twoSum([2, 7, 11, 15], 9)); // [0, 1]

反转字符串 ​

编写一个函数，其作用是将输入的字符串反转过来。

方法1：数组反转 ​

javascript
function reverseString(s) {
    return s.split('').reverse().join('');
}
// 时间复杂度：O(n) 空间复杂度：O(n)

方法2：双指针 ​

javascript
function reverseString(s) {
    const arr = s.split('');
    let left = 0;
    let right = arr.length - 1;
    
    while (left < right) {
        [arr[left], arr[right]] = [arr[right], arr[left]];
        left++;
        right--;
    }
    
    return arr.join('');
}
// 时间复杂度：O(n) 空间复杂度：O(n)

// 测试
console.log(reverseString('hello')); // 'olleh'

function isPalindrome(s) {
    const cleaned = s.toLowerCase().replace(/[^a-z0-9]/g, '');

最长公共前缀 ​

编写一个函数来查找字符串数组中的最长公共前缀。如果不存在公共前缀，返回空字符串。

方法1：横向扫描 ​

javascript
function longestCommonPrefix(strs) {
    if (strs.length === 0) return '';
    
    let prefix = strs[0];
    
    for (let i = 1; i < strs.length; i++) {
        while (strs[i].indexOf(prefix) !== 0) {
            prefix = prefix.slice(0, -1);
            if (prefix === '') return '';
        }
    }
    
    return prefix;
}
// 时间复杂度：O(S)，S 是所有字符串字符数之和

// 测试
console.log(longestCommonPrefix(['flower', 'flow', 'flight'])); // 'fl'

方法2：纵向扫描 ​

javascript
function longestCommonPrefix(strs) {
    if (strs.length === 0) return '';
    
    for (let i = 0; i < strs[0].length; i++) {
        const char = strs[0][i];
        for (let j = 1; j < strs.length; j++) {
            if (i === strs[j].length || strs[j][i] !== char) {
                return strs[0].slice(0, i);
            }
        }
    }
    
    return strs[0];
}

合并两个有序数组 ​

给你两个按非递减顺序排列的整数数组 nums1 和 nums2，请你合并 nums2 到 nums1 中，使合并后的数组同样按非递减顺序排列。

方法：从后往前合并 ​

javascript
function merge(nums1, m, nums2, n) {
    let i = m - 1;      // nums1 有效元素末尾
    let j = n - 1;      // nums2 末尾
    let k = m + n - 1;  // 合并后数组末尾
    
    while (j >= 0) {
        if (i >= 0 && nums1[i] > nums2[j]) {
            nums1[k--] = nums1[i--];
        } else {
            nums1[k--] = nums2[j--];
        }
    }
}
// 时间复杂度：O(m + n) 空间复杂度：O(1)

// 测试
const nums1 = [1, 2, 3, 0, 0, 0];
const nums2 = [2, 5, 6];
merge(nums1, 3, nums2, 3);
console.log(nums1); // [1, 2, 2, 3, 5, 6]

删除排序数组中的重复项 ​

给你一个升序排列的数组 nums，请你原地删除重复出现的元素，使每个元素只出现一次，返回删除后数组的新长度。

方法：双指针 ​

javascript
function removeDuplicates(nums) {
    if (nums.length === 0) return 0;
    
    let slow = 0;
    for (let fast = 1; fast < nums.length; fast++) {
        if (nums[fast] !== nums[slow]) {
            slow++;
            nums[slow] = nums[fast];
        }
    }
    
    return slow + 1;
}
// 时间复杂度：O(n) 空间复杂度：O(1)

// 测试
const nums = [1, 1, 2];
const len = removeDuplicates(nums);
console.log(len, nums.slice(0, len)); // 2, [1, 2]

最大子数组和 ​

给你一个整数数组 nums，请你找出一个具有最大和的连续子数组（子数组最少包含一个元素），返回其最大和。

方法：动态规划（Kadane算法） ​

javascript
function maxSubArray(nums) {
    let maxSum = nums[0];
    let currentSum = nums[0];
    
    for (let i = 1; i < nums.length; i++) {
        currentSum = Math.max(nums[i], currentSum + nums[i]);
        maxSum = Math.max(maxSum, currentSum);
    }
    
    return maxSum;
}
// 时间复杂度：O(n) 空间复杂度：O(1)

// 测试
console.log(maxSubArray([-2, 1, -3, 4, -1, 2, 1, -5, 4])); // 6

有效的括号 ​

给定一个只包括 '('，')'，'{'，'}'，'['，']' 的字符串，判断字符串是否有效。

方法：栈 ​

javascript
function isValid(s) {
    const stack = [];
    const map = {
        ')': '(',
        '}': '{',
        ']': '['
    };
    
    for (let char of s) {
        if (char in map) {
            // 右括号
            if (stack.length === 0 || stack.pop() !== map[char]) {
                return false;
            }
        } else {
            // 左括号
            stack.push(char);
        }
    }
    
    return stack.length === 0;
}
// 时间复杂度：O(n) 空间复杂度：O(n)

// 测试
console.log(isValid('()[]{}'));  // true
console.log(isValid('([)]'));    // false

合并区间 ​

以数组 intervals 表示若干个区间的集合，其中单个区间为 intervals[i] = [starti, endi]。请你合并所有重叠的区间，并返回一个不重叠的区间数组。

方法：排序 + 合并 ​

javascript
function mergeIntervals(intervals) {
    if (intervals.length === 0) return [];
    
    // 按起始位置排序
    intervals.sort((a, b) => a[0] - b[0]);
    
    const merged = [intervals[0]];
    
    for (let i = 1; i < intervals.length; i++) {
        const current = intervals[i];
        const last = merged[merged.length - 1];
        
        if (current[0] <= last[1]) {
            // 重叠，合并
            last[1] = Math.max(last[1], current[1]);
        } else {
            // 不重叠，添加新区间
            merged.push(current);
        }
    }
    
    return merged;
}
// 时间复杂度：O(n log n) 空间复杂度：O(n)

// 测试
console.log(mergeIntervals([[1,3],[2,6],[8,10],[15,18]]));
// [[1,6],[8,10],[15,18]]

买卖股票的最佳时机 ​

给定一个数组 prices，它的第 i 个元素 prices[i] 表示一支给定股票第 i 天的价格。你只能选择某一天买入这只股票，并选择在未来的某一个不同的日子卖出该股票。设计一个算法来计算你所能获取的最大利润。

方法：一次遍历 ​

javascript
function maxProfit(prices) {
    let minPrice = Infinity;
    let maxProfit = 0;
    
    for (let price of prices) {
        minPrice = Math.min(minPrice, price);
        maxProfit = Math.max(maxProfit, price - minPrice);
    }
    
    return maxProfit;
}
// 时间复杂度：O(n) 空间复杂度：O(1)

// 测试
console.log(maxProfit([7, 1, 5, 3, 6, 4])); // 5（第2天买入，第5天卖出）

二分查找 ​

给定一个 n 个元素有序的（升序）整型数组 nums 和一个目标值 target，写一个函数搜索 nums 中的 target，如果目标值存在返回下标，否则返回 -1。

方法：二分查找 ​

javascript
function binarySearch(nums, target) {
    let left = 0;
    let right = nums.length - 1;
    
    while (left <= right) {
        const mid = Math.floor((left + right) / 2);
        
        if (nums[mid] === target) {
            return mid;
        } else if (nums[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }
    
    return -1;
}
// 时间复杂度：O(log n) 空间复杂度：O(1)

// 测试
console.log(binarySearch([-1, 0, 3, 5, 9, 12], 9)); // 4
