环境速查 ​

bash
# 查看版本
python3 --version

# 创建并启用虚拟环境（项目根目录）
python3 -m venv .venv
source .venv/bin/activate   # macOS / Linux
# .venv\Scripts\activate    # Windows

# 安装依赖
pip install 包名
pip install -r requirements.txt

# 退出虚拟环境
deactivate
